func calculate_entropy(probability_distribution):
    # Shannon entropy
    H = 0
    for p in probability_distribution:
        if p > 0:
            H -= p * log2(p)
    return H

# Maximum entropy = uniform distribution
# All states equally possible = maximum freedom

# Minimum entropy = single state
# Only one possibility = no freedom
