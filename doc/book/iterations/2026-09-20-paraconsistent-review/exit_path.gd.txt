
func exit_path(hall: Node3D, origin: Vector3) -> Array:
	# A finite collision survey of this flat hall, followed by actual walking below.
	# Capsule clearance alone is not evidence that a CharacterBody can follow it.
	var space: PhysicsDirectSpaceState3D=hall.get_world_3d().direct_space_state
	var capsule:=CapsuleShape3D.new()
	capsule.radius=.34
	capsule.height=1.8
	var query:=PhysicsShapeQueryParameters3D.new()
	query.shape=capsule
	query.collision_mask=1
	var open: Dictionary={}
	for x in range(3,48):
		for z in range(19,47):
			query.transform=Transform3D(Basis.IDENTITY,origin+Vector3(float(x)*.25,.95,float(z)*.25))
			if space.intersect_shape(query,1).is_empty(): open[Vector2i(x,z)]=true
	var start:=Vector2i(15,21)
	var goal:=Vector2i(26,46)
	var queue: Array[Vector2i]=[start]
	var parent: Dictionary={start:start}
	var cursor: int=0
	while cursor<queue.size():
		var at: Vector2i=queue[cursor]
		cursor+=1
		if at==goal: break
		for dir in [Vector2i.RIGHT,Vector2i.DOWN,Vector2i.LEFT,Vector2i.UP]:
			var next: Vector2i=at+dir
			if open.has(next) and not parent.has(next):
				parent[next]=at
				queue.append(next)
	if not parent.has(goal): return []
	var path: Array[Vector2i]=[goal]
	while path[-1]!=start: path.append(parent[path[-1]])
	path.reverse()
	# Keep turns and coalesce straight sections up to one metre.
	var selected: Array=[Vector3(3.8,.1,5.3)]
	var previous_dir:=Vector2i.ZERO
	var last_kept:=start
	for i in range(1,path.size()):
		var dir: Vector2i=path[i]-path[i-1]
		if dir!=previous_dir or path[i].distance_to(last_kept)>3:
			var at: Vector2i=path[i-1]
			selected.append(Vector3(at.x*.25,.1,at.y*.25))
			last_kept=at
		previous_dir=dir
	selected.append(Vector3(goal.x*.25,.1,goal.y*.25))
	return selected
