Question: Can I move a tree's endpoints while preserving its fork openings and even the final twigs' directions? I choose a periodic influence rather than a random field because I can state the relationships before seeing the silhouette. I want a shared bend by generation, not independently jittered branches or a moving tree.

Input recipe: generations = 3, ratio = 0.7, amplitude 12 degrees, frequency 0.25 cycles per parent-depth unit, phase PI / 2. Parent depth 0 is the trunk, depth 1 its children, and so on. Both siblings at each fork receive the same offset; branches at the same depth share a sample. The seed trunk remains length 0.7 at station position (0, 0.60, 0).

Receiver: only the construction assignment to child.rotation.z: TURN * sign_value + fork_bias(depth). fork_bias returns the wave sample in radians, with amplitude bounded to 0–15 degrees. The copied drawing and first-fork grip code remain. The original stays in reconstructed/Fractal_RecursiveTrees/.

Endpoints should move, while the initial tree remains planar. Expected invariants: 15 segments; lengths 0.7, 0.49, 0.343 and 0.2401 metres in populations 1, 2, 4 and 8; two children per nonterminal branch; each child origin at its parent's tip; 70-degree angle between a fork's children at construction; finite transforms and unit scale. Controls for the comparison: amplitude zero should reconstruct the original transforms, modulo floating-point tolerance. func _ready() -> void:
    assert(is_finite(amplitude_degrees) and is_finite(cycles_per_level) and is_finite(phase))
    tree = branch(0.7, clampi(generations, 1, 4))
    tree.position = Vector3(0, 0.60, 0)
    add_child(tree)
    for child in tree.get_children():
        if child.has_meta("length"):
            first_forks.append(child)
    for i in range(2):
        var point = PointBody.new()
        point.tint = Color("e99db8") if i == 0 else Color("25212a")
        add_child(point)
        points.append(point)
        park_point(i)

func fork_bias(depth: int) -> float:
    var amplitude := deg_to_rad(clampf(amplitude_degrees, 0.0, 15.0))
    var frequency := clampf(cycles_per_level, 0.0, 1.0)
    return amplitude * sin(TAU * frequency * float(depth) + phase)

func branch(length: float, remaining: int, depth: int = 0) -> Node3D:
    var joint := Node3D.new()
    joint.set_meta("length", length)
    var line := LineSkin.new()
    joint.add_child(line)
    line.show_segment(Vector3.ZERO, Vector3.UP * length, length)
    segment_count += 1
    if remaining <= 0:
        return joint
    for sign_value in [-1.0, 1.0]:
        var child := branch(length * clampf(ratio, 0.4, 0.8), remaining - 1, depth + 1)
        child.position = Vector3.UP * length
        child.rotation.z = TURN * sign_value + fork_bias(depth)
        joint.add_child(child)
    return joint

func _process(_delta: float) -> void:
    for i in range(2):
        if points[i].held():
            var fork := first_forks[i]
            var parent: Node3D = fork.get_parent()
            var displacement := parent.to_local(points[i].global_position) - fork.position
            turn_fork(i, displacement)
        else:
            park_point(i)

func turn_fork(index: int, displacement: Vector3) -> void:
    if not displacement.is_finite() or displacement.length() < 0.001:
        return
    var horizontal := Vector3(displacement.x, 0, displacement.z)
    if horizontal.length() < 0.001:
        return
    var angle := clampf(atan2(horizontal.length(), displacement.y), deg_to_rad(12), deg_to_rad(65))
    var direction := Vector3.UP * cos(angle) + horizontal.normalized() * sin(angle)
    first_forks[index].basis = Basis(Quaternion(Vector3.UP, direction))

func park_point(index: int) -> void:
    var fork := first_forks[index]
    var tip := Vector3.UP * float(fork.get_meta("length"))
    points[index].global_position = fork.to_global(tip)
