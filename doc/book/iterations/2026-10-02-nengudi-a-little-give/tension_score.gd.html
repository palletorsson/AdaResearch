extends RefCounted
## Small discrete membrane, solved at a fixed step with compliant distance links.
## Metres / seconds / kilograms. An educational model, not measured nylon.
const NX := 25
const NZ := 9
const STEP := 1.0 / 120.0
const NODE_MASS := 0.035
const LOAD_KG := [0.0, 1.0, 3.0]
var compliance := 0.0015
var damping := 2.8
var load_kg := 0.0
var points := PackedVector3Array()
var previous := PackedVector3Array()
var rest := PackedVector3Array()
var links: Array[Vector2i] = []
var lengths := PackedFloat32Array()
var pins := PackedInt32Array([0, NX-1, (NZ-1)*NX, NZ*NX-1])
var grip_index := (NZ-1)*NX + NX/2
var load_indices := PackedInt32Array()
var grip_held := false
var grip_target := Vector3.ZERO

func _init() -> void:
	for z in NZ:
		for x in NX:
			var u := float(x)/(NX-1)
			# A narrow waist gives the sheet four tapering reaches toward its pins.
			var width := lerpf(0.30,1.1,pow(absf(u*2.0-1.0),1.6))
			rest.append(Vector3(lerpf(-2.8,2.8,u),2.25,lerpf(-width,width,float(z)/(NZ-1))))
			if x>0: add_link(z*NX+x-1,z*NX+x)
			if z>0: add_link((z-1)*NX+x,z*NX+x)
			if z>0 and x>0:
				add_link((z-1)*NX+x-1,z*NX+x)
				add_link((z-1)*NX+x,z*NX+x-1)
	for z in range(3,6):
		for x in range(11,14): load_indices.append(z*NX+x)
	reset()

func add_link(a: int,b: int) -> void:
	links.append(Vector2i(a,b)); lengths.append(rest[a].distance_to(rest[b]))

func reset() -> void:
	points = rest.duplicate(); previous = rest.duplicate()
	load_kg = 0.0; grip_held = false

func bounded_target(target: Vector3) -> Vector3:
	return Vector3(clampf(target.x,-2.2,2.2),clampf(target.y,0.45,3.6),clampf(target.z,-1.8,2.4))

func inverse_mass(index: int) -> float:
	if index in pins or (grip_held and index==grip_index): return 0.0
	return 1.0/NODE_MASS

func step() -> void:
	var retention := exp(-damping*STEP)
	for i in points.size():
		if i in pins: points[i]=rest[i]; previous[i]=rest[i]; continue
		if grip_held and i==grip_index:
			points[i]=bounded_target(grip_target); previous[i]=points[i]; continue
		var acceleration := Vector3(0,-9.81,0)
		if i in load_indices: acceleration.y -= 9.81*load_kg/(load_indices.size()*NODE_MASS)
		var at := points[i]
		points[i] += (points[i]-previous[i])*retention + acceleration*STEP*STEP
		previous[i]=at
	var lambdas := PackedFloat32Array(); lambdas.resize(links.size())
	var alpha := compliance/(STEP*STEP)
	for iteration in 8:
		for n in links.size():
			var a := links[n].x; var b := links[n].y
			var delta := points[b]-points[a]; var distance := delta.length()
			var wa := inverse_mass(a); var wb := inverse_mass(b)
			if distance<0.00001 or wa+wb==0.0: continue
			var dl := (-(distance-lengths[n])-alpha*lambdas[n])/(wa+wb+alpha)
			lambdas[n] += dl
			var direction := delta/distance
			points[a] -= wa*dl*direction
			points[b] += wb*dl*direction
		for i in points.size():
			if points[i].y < 0.12: points[i].y=0.12; previous[i].y=0.12

func load_position() -> Vector3:
	var total := Vector3.ZERO
	for i in load_indices: total += points[i]
	return total/float(load_indices.size())
